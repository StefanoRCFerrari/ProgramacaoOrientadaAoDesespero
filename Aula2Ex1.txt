package estudosp1;
//A2Ex1
import java.util.Scanner;

public class EstudosP1 {

    public static void main(String[] args) {
        
        Scanner scanner = new Scanner(System.in);
        int[] vetor = new int[10];
        int num, aux;
        boolean baguncado = true;
        
        for (int contador = 0; contador < 10; contador++){
            System.out.println("Insira o " + (contador+1) + "º número:  ");
            num = scanner.nextInt();
            vetor[contador] = num;
        }
        
        while (baguncado){
            baguncado = false;
            for (int contador = 0; contador < 9; contador++){
                if (vetor[contador] > vetor[contador+1]){
                    aux = vetor[contador];
                    vetor[contador] = vetor[contador+1];
                    vetor[contador+1] = aux;
                    baguncado = true;
                }
            }
        }
        
        System.out.println("Iniciando print ordenado: ");
        
        for (int contador = 0; contador < 10; contador++){
            System.out.print(vetor[contador] + " ");
        }
    }
    
}
