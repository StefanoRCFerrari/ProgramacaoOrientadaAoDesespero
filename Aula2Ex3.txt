package estudosp1;
//A2Ex3
import java.util.Scanner;

public class EstudosP1 {

    public static void main(String[] args) {
        
        Scanner scanner = new Scanner(System.in);
        
        int[][] matriz1 = new int[3][3];
        int[][] matriz2 = new int[3][3];
        int[][] matriz3 = new int[3][3];
        int contador, contador2, contador3, elemento = 0;
        
        for (contador = 0; contador < 3; contador++){
            for (contador2= 0; contador2 < 3; contador2++){
                System.out.println("Elemento da posicao " + (contador+1) + "x" + (contador2+1) + " da matriz 1: ");
                matriz1[contador][contador2] = scanner.nextInt();
            }
        }
        
        for (contador = 0; contador < 3; contador++){
            for (contador2= 0; contador2 < 3; contador2++){
                System.out.println("Elemento da posicao " + (contador+1) + "x" + (contador2+1) + " da matriz 2: ");
                matriz2[contador][contador2] = scanner.nextInt();
            }
        }
        
        System.out.println("Matriz 1: ");
        for (contador = 0; contador < 3; contador++){
            System.out.print("\n");
            for (contador2= 0; contador2 < 3; contador2++){
                System.out.print(matriz1[contador][contador2] + " ");
            }
        }    
        
        System.out.print("\n");
        
        System.out.println("Matriz 2: ");
        for (contador = 0; contador < 3; contador++){
            System.out.print("\n");
            for (contador2= 0; contador2 < 3; contador2++){
                System.out.print(matriz2[contador][contador2] + " ");
            }
        }   
    
    
        for (contador = 0; contador < 3; contador++){
            for (contador2 = 0; contador2 < 3; contador2++){
                for (contador3 = 0; contador3 < 3; contador3++){
                    elemento += (matriz1[contador][contador3] * matriz2[contador3][contador2]);
                }
            matriz3[contador][contador2] = elemento;
            elemento = 0;
            }
        }
        
        System.out.print("\n");
        
        System.out.println("Matriz 3: ");
        for (contador = 0; contador < 3; contador++){
            System.out.print("\n");
            for (contador2= 0; contador2 < 3; contador2++){
                System.out.print(matriz3[contador][contador2] + " ");
            }
        }   
    }
}
